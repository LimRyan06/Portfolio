AlzCare

The problem

Current dementia care apps are heavily fragmented and static. They fail to adapt to a patient's cognitive decline over time, and they don't provide caregivers with proactive, data-driven insights to prevent medication non-adherence before it happens.

The solution

An end-to-end platform featuring a dual-interface mobile app (Flutter) synced in real-time via a Python backend. It utilizes a GRU Neural Network to analyze time-series data and predict medication non-adherence risks, alongside Contextual Bandits that dynamically adjust cognitive game difficulty in real-time based on patient performance.

gru_model.py 
# GRU risk prediction for patients medication adherence 
# pipeline: input -> GRU layer -> linear layer -> sigmoid -> risk score (0-1)

import torch    # library for deep learning operations
import torch.nn as nn    # neural network module components

# GRU class
class AdherenceGRU(nn.Module):
    def __init__(
        self,
        input_size: int = 1,    # daily features (1.0 is taken, rest is 0.0)
        hidden_size: int = 50,    # hidden GRU units size
        num_layers: int = 1,    # stacked GRU layers 
        dropout: float = 0.0,    # dropout between stacked layers (active when theres more than 1 layer in num_layers)
    ):
        super(AdherenceGRU, self).__init__()

        # sets hidden unit and layer count parameters
        self.hidden_size = hidden_size 
        self.num_layers = num_layers

        # GRU layer
        self.gru = nn.GRU(
            input_size=input_size,
            hidden_size=hidden_size,
            num_layers=num_layers,
            batch_first=True,     # allows standard input channel order (batch, sequence, feature)
            dropout=dropout if num_layers > 1 else 0.0,
        )

        self.fc = nn.Linear(hidden_size, 1)     # Fully connected output layer
        self.sigmoid = nn.Sigmoid()     # Sigmoid to force output to [0, 1]

    # performs GRU model forward pass computation (x is torch.Tensor, shape (batch, sequence length, input size))
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        out, _ = self.gru(x)     # passes inputs through active GRU cells
        last_hidden = out[:, -1, :]   # extracts summary of the last day in the sequence
        logit = self.fc(last_hidden)  # calculates raw score
        risk   = self.sigmoid(logit)  # convert raw score to risk percentage
        return risk

GRU risk-prediction architecture

A PyTorch nn.GRU processes sequential adherence inputs, uses the final sequence representation and passes it through a linear layer plus sigmoid to return a non-adherence risk score between 0 and 1.

train_gru.py 
# training script for GRU (fetches all med logs across all patients -> builds a dataset of sequences with labels -> trains GRU for 50 epochs using Adam + BCELoss)

import torch    # library for deep learning operations
import torch.nn as nn    # neural network module components
from torch.utils.data import DataLoader, TensorDataset    # dataloader to manage batching and shuffling, tensordataset to wrap tensors
from ml.gru_model import AdherenceGRU     # imports GRU model
from services.firebase import initialize_firebase, get_db      # connects to firebase

# hyper parameters
SEQ_LEN    = 14
HIDDEN     = 50
EPOCHS     = 50
BATCH_SIZE = 8
LR         = 0.001
WEIGHTS_OUT = "ml/gru_weights.pth"
MIN_SAMPLES = 10   # minimum sequences required for training

# pulls all adherence logs and creates sliding window sequences
def fetch_training_data() -> tuple[list, list]:
    db = get_db()
    patient_docs = list(db.collection("patients").stream())

    X: list[list[float]] = []    # features (sliding window)
    Y: list[float] = []    # labels (binary risk 1.0 - 0.0)

    # fetch data
    for patient_doc in patient_docs:
        patient_id = patient_doc.id
        docs = (
            db.collection("adherence_logs")
            .where("patient_id", "==", patient_id)
            .stream()
        )
        # sort in memory to avoid composite index requirement from Firestore (avoid crashing from missing index)
        raw = [doc.to_dict() for doc in docs]
        raw.sort(key=lambda d: d.get("timestamp", ""))
        events: list[float] = []
        for entry in raw:
            status = entry.get("status", "missed")
            events.append(1.0 if status == "taken" else 0.0)
        # needs at least 1 seq_len and 1 event to form a training sample
        if len(events) < SEQ_LEN + 1:
            continue
        # sliding window (each window predicts next event)
        for i in range(len(events) - SEQ_LEN):
            seq   = events[i : i + SEQ_LEN]
            label = events[i + SEQ_LEN]
            risk_label = 1.0 - label
            X.append(seq)
            Y.append(risk_label)

    return X, Y

def train() -> None:
    initialize_firebase()
    print("[Train] Connected to Firestore ")
    X, Y = fetch_training_data()
    print(f"[Train] Collected {len(X)} training sequences from Firestore.")
    if len(X) < MIN_SAMPLES:
        print(
            f"[Train] Not enough data yet ({len(X)} sequences, need {MIN_SAMPLES}+). "
            "Log more medication events in the app and re-run training."
        )
        return

    # Build tensors for pytorch model (shape: (N, SEQ_LEN, 1) and (N, 1))
    X_tensor = torch.tensor(X, dtype=torch.float32).unsqueeze(-1)
    Y_tensor = torch.tensor(Y, dtype=torch.float32).unsqueeze(-1)
    dataset = TensorDataset(X_tensor, Y_tensor)
    loader  = DataLoader(dataset, batch_size=BATCH_SIZE, shuffle=True)
    model     = AdherenceGRU(input_size=1, hidden_size=HIDDEN, num_layers=1)
    criterion = nn.BCELoss()
    optimizer = torch.optim.Adam(model.parameters(), lr=LR)
    model.train()
    print(f"[Train] Training for {EPOCHS} epochs with batch_size={BATCH_SIZE}, lr={LR}…\n")

    # train model
    for epoch in range(1, EPOCHS + 1):
        epoch_loss = 0.0
        for batch_X, batch_Y in loader:
            optimizer.zero_grad()
            output = model(batch_X)
            loss   = criterion(output, batch_Y)
            loss.backward()
            optimizer.step()
            epoch_loss += loss.item()
        # calculate the average loss (dividing total loss by number of batches)
        avg_loss = epoch_loss / len(loader)
        if epoch % 10 == 0 or epoch == 1:
            print(f"  Epoch {epoch:>3}/{EPOCHS}  -  BCELoss: {avg_loss:.4f}")
    # save model weights
    torch.save(model.state_dict(), WEIGHTS_OUT)
    print(f"\n[Train] Model weights saved to {WEIGHTS_OUT}")
    print("[Train] Restart the API server to load the updated weights.")

if __name__ == "__main__":
    train()

GRU training pipeline

The training service retrieves adherence logs from Firestore, creates 14-day sliding-window sequences, batches them with a DataLoader, optimises binary cross-entropy loss with Adam and saves the trained PyTorch weights for the API.

contextual_bandit.py 
# Contextual bandit (Epsilon-greedy algorithm) for difficulty recommendation
# every patient's state is maintained in the firestore 'bandit_state' collection so recommendations can bypass server restarts and improve over time

import random     # random variable generator
from services.firebase import get_db     # helper to retrieve active firestore database

# difficulty options
ARMS = ["easy", "medium", "hard"]
EPSILON = 0.2   # 20% explore, 80% exploit
# difficulty benchmarks used for reward calculations
_BENCHMARKS = {
    "easy":   {"max_errors": 6,  "time_par": 45},
    "medium": {"max_errors": 10, "time_par": 90},
    "hard":   {"max_errors": 15, "time_par": 150},
}
# return initial default state values
def _default_state() -> dict:
    return {
        "counts":   {arm: 0   for arm in ARMS},
        "q_values": {arm: 0.0 for arm in ARMS},
    }
# loads and validates current bandit state from firestore
def _load_state(patient_id: str) -> dict:
    db  = get_db()
    doc = db.collection("bandit_state").document(patient_id).get()
    if doc.exists:
        data = doc.to_dict()
        # make sure all arms present (in case for change of schema)
        counts   = {arm: data.get("counts",   {}).get(arm, 0)   for arm in ARMS}
        q_values = {arm: data.get("q_values", {}).get(arm, 0.0) for arm in ARMS}
        return {"counts": counts, "q_values": q_values}
    return _default_state()

# saves latest updated model state to firestore
def _save_state(patient_id: str, state: dict) -> None:
    db = get_db()
    db.collection("bandit_state").document(patient_id).set(state, merge=True)
 
# chooses recommendation using epsilon greedy exploration
def get_recommendation(patient_id: str) -> dict:
    state = _load_state(patient_id)
    q     = state["q_values"]
    if random.random() < EPSILON:
        chosen = random.choice(ARMS)
        method = "explore"
    else:
        chosen = max(q, key=q.get)
        method = "exploit"
    return {
        "recommended_difficulty": chosen,
        "method":   method,
        "q_values": q,
        "counts":   state["counts"],
    }

# calculate game reward and update q-values in firestore
def update_bandit(
    patient_id: str,
    difficulty: str,
    score: int,
    completion_time: float,
    errors: int,
) -> float:
    if difficulty not in ARMS:
        return 0.0
    bench      = _BENCHMARKS[difficulty]
    max_errors = bench["max_errors"]
    time_par   = bench["time_par"]

    # base reward for completing game
    base_reward = 0.5
    # accuracy bonus (lesser errors give higher rewards)
    acc_bonus = 0.3 * max(0.0, 1.0 - errors / max(max_errors, 1))
    # speed bonus (faster times give higher rewards)
    speed_bonus = 0.2 * max(0.0, min(1.0, 1.0 - completion_time / max(time_par, 1)))
    reward = round(base_reward + acc_bonus + speed_bonus, 4)
    # incremental running average update 
    state = _load_state(patient_id)
    n     = state["counts"][difficulty] + 1
    old_q = state["q_values"][difficulty]
    new_q = old_q + (reward - old_q) / n   # running average

    state["counts"][difficulty]   = n
    state["q_values"][difficulty] = round(new_q, 6)
    _save_state(patient_id, state)
    return reward

Adaptive game-difficulty recommender

The epsilon-greedy contextual bandit balances exploration and exploitation across easy, medium and hard game difficulties. Each patient's Q-values and selection counts are persisted in Firestore, while rewards combine completion, accuracy and speed so recommendations can adapt as the patient continues using AlzCare.

Cambridge Battlecode — Team 0_day

A storyboard-style recap of building and iterating autonomous bots under competition pressure, tracking our progress on the live ladder and refining strategy round by round.

What this experience was about

Cambridge Battlecode is a six-week programming competition where teams build bots to battle in a turn-based strategy game. Our team, 0_day, competed in the UK Main category and focused on building resource-harvesting, exploration and conveyor-network strategies.

What made the experience especially valuable was the pace of iteration: every improvement had to be tested quickly, observed through match history and visualisers, and then refined again as the competition evolved.
